%% quotes (requires graphicx)
\bragiProvideCommand{\OpenQuote}{\scalebox{1.5}{``}}
\bragiProvideCommand{\CloseQuote}{\scalebox{1.5}{''}}
\bragiProvideCommand{\QuoteBy}[2]{\OpenQuote \textit{#1} \CloseQuote\newline --- #2}

\bragiProvideCommand{\centeronpage}[1]{%
      \noindent
      \hspace*{-0.5\dimexpr\paperwidth-\textwidth\relax}%
      \makebox[\paperwidth][c]{#1}%
}
