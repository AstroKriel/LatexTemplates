\bragiProvideCommand{\mRem}{\ensuremath{\mathrm{Rm}}} % magnetic Reynolds number
\bragiProvideCommand{\mPrm}{\ensuremath{\mathrm{Pm}}} % magnetic Prandtl number: Ohmic vs shear
\bragiProvideCommand{\mPrnu}{\ensuremath{\mathrm{P}\nu}} % viscous Prandtl number: shear vs bulk
\bragiProvideCommand{\mSonicMach}{\ensuremath{\mathcal{M}}} % sonic Mach number
\bragiProvideCommand{\mAlfvenMach}{\ensuremath{\mathcal{M}_{\mathrm{A}}}} % Alfvenic Mach number
\bragiProvideCommand{\mPlasmaBeta}{\ensuremath{\beta}} % plasma-beta: gas vs magnetic pressure

%% calligraphic quantities
\bragiProvideCommand{\mSpectrum}{\ensuremath{\mathcal{P}}} % power spectrum
\bragiProvideCommand{\mVolume}{\ensuremath{\mathcal{V}}} % volume
\bragiProvideCommand{\mSurface}{\ensuremath{\partial\mathcal{V}}} % boundary surface of a volume
