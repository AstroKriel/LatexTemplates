%% vector calculus operators
\bragiProvideCommand{\mDiv}[1]{\nabla \cdot #1} % divergence
\bragiProvideCommand{\mCurl}[1]{\nabla \times #1} % curl
\bragiProvideCommand{\mGrad}[1]{\nabla #1} % gradient
\bragiProvideCommand{\mGradPerp}[1]{\nabla_\perp #1} % gradient perpendicular to a reference direction
\bragiProvideCommand{\mLaplace}[1]{\Delta #1} % Laplacian (for scalar fields)

%% vectors / tensors
\bragiProvideCommand{\mUnit}[1]{\widehat{#1}}
\bragiProvideCommand{\mVector}[1]{\bm{#1}} % requires bm
\bragiProvideCommand{\mVectorUnit}[1]{\mUnit{\mVector{#1}}} % unit vector
\bragiProvideCommand{\mVectorBase}[1]{\hat{\mVector{e}}_{#1}} % basis vector
\bragiProvideCommand{\mTensor}[1]{\mathbb{#1}} % tensor font
\bragiProvideCommand{\imatrix}{\mTensor{I}} % identity matrix

%% differentials
\bragiProvideCommand{\mdd}[1]{\mathrm{d} #1}
\bragiProvideCommand{\mpp}[1]{\partial #1}
\bragiProvideCommand{\mDD}[1]{\mathrm{D} #1}

%% derivatives in fractional form
\bragiProvideCommand{\mddFrac}[2]{\frac{\mdd{#1}}{\mdd{#2}}}
\bragiProvideCommand{\mppFrac}[2]{\frac{\mpp{#1}}{\mpp{#2}}}
\bragiProvideCommand{\mDDFrac}[2]{\frac{\mDD{#1}}{\mDD{#2}}}
