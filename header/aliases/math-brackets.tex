%% brackets
\bragiProvideCommand{\rbrac}[1]{\left( #1 \right)} % round brackets
\bragiProvideCommand{\sbrac}[1]{\left[ #1 \right]} % square brackets
\bragiProvideCommand{\cbrac}[1]{\left\{ #1 \right\}} % curly brackets
\bragiProvideCommand{\nbrac}[1]{\left\| #1 \right\|} % norm brackets

\bragiProvideCommand{\rfrac}[2]{\left( \frac{#1}{#2} \right)} % parenthesised fraction

%% operators
\bragiProvideCommand{\mAbs}[1]{\left| #1 \right|} % absolute
\bragiProvideCommand{\mAve}[1]{\left\langle #1 \right\rangle} % average
\bragiProvideCommand{\mCeil}[1]{\left\lceil #1 \right\rceil} % ceiling
\bragiProvideCommand{\mFloor}[1]{\left\lfloor #1 \right\rfloor} % floor

%% asymptotics
\bragiProvideCommand{\mBigO}[1]{\mathcal{O}\left[ #1 \right]} % big-O notation
