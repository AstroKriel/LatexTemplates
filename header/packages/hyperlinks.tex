%% load hyperref only if the document class hasn't already (many journal classes,
%% e.g. mnras.cls, load it themselves); bookmark and cleveref just need to come after it,
%% which this input order guarantees.
\makeatletter
\@ifpackageloaded{hyperref}{}{\usepackage{hyperref}}
\@ifpackageloaded{bookmark}{}{\usepackage{bookmark}} % for: more reliable PDF bookmarks/outline than hyperref alone
\makeatother

\hypersetup{
  colorlinks=true,
  linkcolor=Blue, % sections, equations, figures (requires dvipsnames, see packages/figures)
  citecolor=Red,  % citations
  urlcolor=Red,   % URLs
}

\makeatletter
\@ifpackageloaded{cleveref}{}{\usepackage[nameinlink]{cleveref}} % for: referencing a range of eqns; nameinlink makes the whole "Section N" clickable
\makeatother
